```bash
#!/bin/bash
# FreeIPA Backup Script — Atius Domain Infrastructure
# Strategy: Stop container → tar /data volume → restart container
# Per research: /data volume backup is the recommended pattern for freeipa-container
# Source: github.com/freeipa/freeipa-container README

set -e

BACKUP_DIR="/var/lib/freeipa/backups"
DATA_DIR="/var/lib/freeipa/data"
DATE=$(date +%Y%m%d-%H%M%S)
RETENTION=7
CONTAINER_NAME="freeipa-server"

mkdir -p "$BACKUP_DIR"

echo "[$(date)] Starting FreeIPA backup..."

# Strategy 1: Volume backup (recommended — stop container for consistent state)
echo "[$(date)] Stopping FreeIPA container for consistent backup..."
docker stop "$CONTAINER_NAME"

echo "[$(date)] Creating backup archive..."
tar czf "$BACKUP_DIR/freeipa-data-${DATE}.tar.gz" -C /var/lib/freeipa data/

echo "[$(date)] Restarting FreeIPA container..."
docker start "$CONTAINER_NAME"

# Wait for container to be healthy (max 60 seconds)
echo "[$(date)] Waiting for FreeIPA to be ready..."
for i in $(seq 1 12); do
  if docker exec "$CONTAINER_NAME" ipa ping >/dev/null 2>&1; then
    echo "[$(date)] FreeIPA is ready."
    break
  fi
  sleep 5
done

# Strategy 2: ipa-backup (native FreeIPA backup — online, no downtime)
# Uncomment below for additional native backup alongside volume backup:
# echo "[$(date)] Creating native ipa-backup..."
# docker exec "$CONTAINER_NAME" ipa-backup --data --online 2>/dev/null || echo "[$(date)] ipa-backup skipped (may require stopped container for full backup)"

# Cleanup old backups beyond retention period
echo "[$(date)] Cleaning up backups older than ${RETENTION} days..."
find "$BACKUP_DIR" -name "freeipa-data-*.tar.gz" -mtime +$RETENTION -delete

# Report
BACKUP_FILE="$BACKUP_DIR/freeipa-data-${DATE}.tar.gz"
BACKUP_SIZE=$(du -h "$BACKUP_FILE" | cut -f1)
echo "[$(date)] Backup complete: $BACKUP_FILE ($BACKUP_SIZE)"
echo "[$(date)] Files in backup directory:"
ls -lh "$BACKUP_DIR"/freeipa-data-*.tar.gz
```

```bash
#!/bin/bash
# FreeIPA Verification Script — Phase 3 Success Criteria
# Tests: FIPA-01 through FIPA-06

set -euo pipefail

PASS=0
FAIL=0
CONTAINER_NAME="freeipa-server"

check() {
  local req_id="$1"
  local description="$2"
  local command="$3"
  
  if eval "$command" >/dev/null 2>&1; then
    echo "[PASS] $req_id: $description"
    PASS=$((PASS + 1))
  else
    echo "[FAIL] $req_id: $description"
    FAIL=$((FAIL + 1))
  fi
}

echo "=== FreeIPA Phase 3 Verification ==="
echo "Date: $(date)"
echo ""

# FIPA-01: Container running healthy
check "FIPA-01" "Container FreeIPA running" \
  "docker ps --filter name=$CONTAINER_NAME --filter status=running --format '{{.ID}}' | grep -q ."

# FIPA-02: CLI accessible
check "FIPA-02" "CLI accessible (ipa user-find)" \
  "docker exec $CONTAINER_NAME ipa user-find admin 2>/dev/null | grep -q 'User name: admin'"

# FIPA-02b: Web UI accessible
check "FIPA-02b" "Web UI accessible (HTTPS 443)" \
  "curl -k -s -o /dev/null -w '%{http_code}' https://10.1.1.1/ipa/ui/ | grep -qE '200|302'"

# FIPA-03: Realm ATIUS.COM.BR
check "FIPA-03" "Realm ATIUS.COM.BR configured" \
  "docker exec $CONTAINER_NAME ipa realm-show 2>/dev/null | grep -q 'ATIUS.COM.BR'"

# FIPA-04: DNS internal resolution
check "FIPA-04" "DNS resolves internal hosts" \
  "docker exec $CONTAINER_NAME host ipa.atius.com.br 2>/dev/null | grep -q 'has address'"

# FIPA-05: CA operational
check "FIPA-05" "CA operational (cert-show)" \
  "docker exec $CONTAINER_NAME ipa cert-show 1 2>/dev/null | grep -q 'Issuer:'"

# FIPA-06: Backup exists
check "FIPA-06" "Backup file exists" \
  "ls /var/lib/freeipa/backups/freeipa-data-*.tar.gz 2>/dev/null | head -1 | xargs test -s"

echo ""
echo "=== Results ==="
echo "Passed: $PASS / $((PASS + FAIL))"
echo "Failed: $FAIL / $((PASS + FAIL))"

if [ $FAIL -gt 0 ]; then
  echo ""
  echo "WARNING: $FAIL verification(s) failed!"
  echo "Check container logs: docker logs $CONTAINER_NAME --tail 50"
  exit 1
else
  echo ""
  echo "All phase 3 success criteria verified."
  exit 0
fi
```
