```markdown
# Tuning Recipes

## Builds (cargo, rustc, gcc)
- cpu.weight: 200 default, 300 se contendido, 150 se ocioso
- io.weight: 200 default
- memory.high: 6G (release builds) / 8G (link)
- Tempo de throttle: 5s acceptable para build

## Transfers (rsync, rclone)
- io.weight: 100 default (lowest)
- cpu.weight: 100 default
- memory.high: 1G (raramente usa mais)
- Pode rodar 24/7 sem impacto

## Generic (yes, find, dd)
- cpu.weight: 25 (mínimo, deprioritizado)
- io.weight: 25
- memory.high: max (legit)
- Catch-all — se rodar muito, reclassificar

## Interactive (obsidian, electron, codex)
- cpu.weight: 500 (priority sob contenção)
- io.weight: 500
- memory.high: 4G
- NUNCA throttle interativos
```

Criar `scripts/horistic-ops.sh`:

```bash
#!/usr/bin/env bash
# horistic-ops.sh — CLI para o agente HoristicOps
# Uso:
#   horistic-ops.sh status         — 22 métricas em JSON
#   horistic-ops.sh diagnose        — top 5 CPU + top 5 MEM + PSI + load
#   horistic-ops.sh forecast        — disk fill em 7d/30d
#   horistic-ops.sh incidents       — events table do DB
#   horistic-ops.sh tune            — show current weights vs baseline
#   horistic-ops.sh help            — usage

set -u
DB=/home/ubuntu/.local/state/omni/mission-guardian.db
log=/home/ubuntu/.logs/resource-governor/horistic-ops.log
mkdir -p "$(dirname "$log")"

case "${1:-help}" in
  status)
    if [ ! -f "$DB" ]; then echo '{"error":"no db"}'; exit 1; fi
    sqlite3 -json "$DB" "SELECT * FROM samples ORDER BY ts DESC LIMIT 1"
    ;;
  diagnose)
    echo "=== top CPU ==="
    ps -eo pid,pcpu,pmem,rss,comm,args --sort=-pcpu | head -6
    echo "=== top MEM ==="
    ps -eo pid,pcpu,pmem,rss,comm,args --sort=-rss | head -6
    echo "=== load ==="
    cat /proc/loadavg
    echo "=== PSI ==="
    for p in cpu memory io; do
      f="/proc/pressure/$p"
      [ -r "$f" ] && { echo "  $p:"; cat "$f" | head -2; }
    done
    echo "=== mem ==="
    free -m | head -2
    echo "=== disk ==="
    df -h / | tail -1
    echo "=== watchdog/patcher ==="
    pgrep -fa resource-governor- | head -2
    echo "=== inviolable watchdog log (last 5) ==="
    tail -5 /home/ubuntu/.logs/resource-governor/inviolable-watchdog.log
    ;;
  forecast)
    if [ ! -f "$DB" ]; then echo "no db yet"; exit 1; fi
    sqlite3 "$DB" "SELECT ts, disk_pct FROM samples ORDER BY ts DESC LIMIT 5"
    echo "---"
    echo "Disk forecast: requires >100 samples (use 'status' to check count)"
    sqlite3 "$DB" "SELECT count(*) || ' samples' FROM samples"
    ;;
  incidents)
    [ -f "$DB" ] && sqlite3 -header -column "$DB" "SELECT ts, level, source, message FROM events ORDER BY ts DESC LIMIT 20" || echo "no db"
    ;;
  tune)
    echo "=== current weights ==="
    for s in omni-builds omni-interactive omni-transfers omni-generic omni-protected; do
      w=$(cat /sys/fs/cgroup/user.slice/user-1001.slice/$s/cpu.weight 2>/dev/null)
      echo "  $s: $w"
    done
    echo "=== baseline ==="
    echo "  omni-builds: 200"
    echo "  omni-interactive: 500"
    echo "  omni-transfers: 100"
    echo "  omni-generic: 25"
    echo "  omni-protected: 1000"
    ;;
  help|*)
    echo "horistic-ops.sh status|diagnose|forecast|incidents|tune|help"
    ;;
esac
```
