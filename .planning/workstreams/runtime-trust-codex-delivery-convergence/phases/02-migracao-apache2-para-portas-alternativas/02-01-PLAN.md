```bash
cat > /tmp/02-cloudflare-rollback.sh << 'ROLLBACK'
#!/bin/bash
# Rollback: restore Cloudflare Origin Rules to pre-migration state
# Run this if something breaks after updating Origin Rules

set -e

if [ -z "$CF_API_TOKEN" ] || [ -z "$CF_ZONE_ID" ]; then
  echo "Error: CF_API_TOKEN and CF_ZONE_ID must be set"
  exit 1
fi

RULESET_ID=$(curl -s "https://api.cloudflare.com/client/v4/zones/$CF_ZONE_ID/rulesets" \
  -H "Authorization: Bearer $CF_API_TOKEN" \
  -H "Content-Type: application/json" | \
  jq -r '.result[] | select(.phase == "http_request_dynamic_routing") | .id')

if [ -z "$RULESET_ID" ] || [ "$RULESET_ID" = "null" ]; then
  echo "No Origin Rules ruleset found — nothing to rollback"
  exit 0
fi

# Restore from backup
if [ -f "/tmp/02-cloudflare-rules-backup.json" ]; then
  echo "Restoring Origin Rules from backup..."
  # Extract just the rules array — backup includes read-only fields that API rejects
  RULES=$(jq '{rules: .result.rules}' /tmp/02-cloudflare-rules-backup.json)
  curl -s -X PUT "https://api.cloudflare.com/client/v4/zones/$CF_ZONE_ID/rulesets/$RULESET_ID" \
    -H "Authorization: Bearer $CF_API_TOKEN" \
    -H "Content-Type: application/json" \
    -d "$RULES" | jq '.success'
  echo "Rollback complete"
else
  echo "Error: No backup file found at /tmp/02-cloudflare-rules-backup.json"
  exit 1
fi
ROLLBACK
chmod +x /tmp/02-cloudflare-rollback.sh
```

**IF no current Origin Rules exist (most likely):** Create a single wildcard rule.

```bash
# Get or create the ruleset
RULESET_ID=$(curl -s "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/rulesets" \
  -H "Authorization: Bearer $CF_API_TOKEN" \
  -H "Content-Type: application/json" | \
  jq -r '.result[] | select(.phase == "http_request_dynamic_routing") | .id' 2>/dev/null)

if [ -z "$RULESET_ID" ] || [ "$RULESET_ID" = "null" ]; then
  # Create new ruleset with wildcard rule
  echo "Creating new Origin Rules ruleset..."
  curl -s -X POST "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/rulesets" \
    -H "Authorization: Bearer $CF_API_TOKEN" \
    -H "Content-Type: application/json" \
    -d '{
      "name": "Origin Rules - route to port 9444",
      "kind": "zone",
      "phase": "http_request_dynamic_routing",
      "rules": [
        {
          "expression": "http.host endswith \".atius.com.br\"",
          "description": "Route all atius.com.br subdomains to origin port 9444",
          "action": "route",
          "action_parameters": {
            "origin": {
              "port": 9444
            }
          }
        },
        {
          "expression": "http.host endswith \".horistic.com\"",
          "description": "Route all horistic.com subdomains to origin port 9444",
          "action": "route",
          "action_parameters": {
            "origin": {
              "port": 9444
            }
          }
        }
      ]
    }' | jq '.'
else
  # Update existing ruleset — GET, modify, PUT
  echo "Updating existing Origin Rules ruleset..."

  # Get current rules
  CURRENT_RULES=$(curl -s "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/rulesets/$RULESET_ID" \
    -H "Authorization: Bearer $CF_API_TOKEN" \
    -H "Content-Type: application/json")

  # Check if wildcard rule already exists
  HAS_WILDCARD=$(echo "$CURRENT_RULES" | jq '.result.rules[] | select(.expression | contains("atius.com.br") or contains("horistic.com"))' 2>/dev/null)

  if [ -n "$HAS_WILDCARD" ]; then
    echo "Wildcard rule already exists — updating port to 9444"
    # Update port in existing wildcard rules
    UPDATED_RULES=$(echo "$CURRENT_RULES" | jq '
      .result.rules |= map(
        if (.expression | contains("atius.com.br") or contains("horistic.com")) then
          .action_parameters.origin.port = 9444
        else .
        end
      )
    ')
  else
    echo "Adding wildcard rule to existing ruleset"
    # Add wildcard rules to existing ruleset
    UPDATED_RULES=$(echo "$CURRENT_RULES" | jq '
      .result.rules += [
        {
          "expression": "http.host endswith \".atius.com.br\"",
          "description": "Route all atius.com.br subdomains to origin port 9444",
          "action": "route",
          "action_parameters": {
            "origin": {
              "port": 9444
            }
          }
        },
        {
          "expression": "http.host endswith \".horistic.com\"",
          "description": "Route all horistic.com subdomains to origin port 9444",
          "action": "route",
          "action_parameters": {
            "origin": {
              "port": 9444
            }
          }
        }
      ]
    ')
  fi

  # PUT updated ruleset back
  echo "$UPDATED_RULES" | jq '{
    rules: .result.rules
  }' | curl -s -X PUT "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/rulesets/$RULESET_ID" \
    -H "Authorization: Bearer $CF_API_TOKEN" \
    -H "Content-Type: application/json" \
    -d @- | jq '.'
fi
```

2. Test each hostname via curl (through Cloudflare):
```bash
for host in "${TEST_HOSTS[@]}"; do
  echo "=== Testing $host ==="
  # HTTPS through Cloudflare (proxied)
  HTTP_CODE=$(curl -sk -o /dev/null -w "%{http_code}" "https://$host/" --connect-timeout 10)
  echo "  HTTPS via Cloudflare: $HTTP_CODE"

  # Check response headers to confirm Cloudflare proxy
  curl -sk -I "https://$host/" --connect-timeout 10 | grep -i "cf-ray\|server: cloudflare" || echo "  No Cloudflare headers (may be DNS-only mode)"

  # If HTTP_CODE is 000, the connection failed (Origin Rules not working or DNS issue)
  if [ "$HTTP_CODE" = "000" ]; then
    echo "  FAILED: Connection refused or timeout"
  elif [ "$HTTP_CODE" -ge 200 ] && [ "$HTTP_CODE" -lt 500 ]; then
    echo "  PASS: Got HTTP $HTTP_CODE"
  else
    echo "  WARNING: Unexpected HTTP $HTTP_CODE"
  fi
  echo ""
done
```
